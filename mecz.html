const canvas = document.getElementById("boxingRing"); const ctx = canvas.getContext("2d"); const commBox = document.getElementById("commentary"); function logEvent(text, color = "#00ffaa") { commBox.innerHTML += `[SYSTEM] ${text}
`; commBox.scrollTop = commBox.scrollHeight; } // Nowoczesne, neonowe definicje zawodników (Wizualny styl Cyber-Glow) let p1 = { x: 180, baseRealY: 280, hp: 100, color: "#ff2255", glowColor: "rgba(255, 34, 85, 0.5)", name: "PROTOTYP_RED", facing: 1, punching: false, punchProgress: 0, punchStage: "out", state: "move", timer: 0, targetX: 180, hitFlash: 0 }; let p2 = { x: 520, baseRealY: 280, hp: 100, color: "#00ccff", glowColor: "rgba(0, 204, 255, 0.5)", name: "PROTOTYP_BLUE", facing: -1, punching: false, punchProgress: 0, punchStage: "out", state: "move", timer: 0, targetX: 520, hitFlash: 0 }; let endOfGame = false; function drawModernFighter(f) { ctx.save(); let groundY = f.baseRealY; // EFEKT HIT-FLASH: Jeśli zawodnik dostał cios, całe jego ciało na chwilę rozbłyskuje na biało let isFlashing = f.hitFlash > 0; if (isFlashing) f.hitFlash--; // Włączenie nowoczesnego efektu poświaty (Glow) dla neonów ctx.shadowBlur = isFlashing ? 30 : 15; ctx.shadowColor = isFlashing ? NOWOCZESNY PANCERZ / KORPUS (Półprzezroczysty poligon z neonową ramką) ctx.fillStyle = "rgba(20, 20, 20, 0.85)"; ctx.strokeStyle = isFlashing ? "#ffffff" : f.color; ctx.lineWidth = 3; ctx.beginPath(); ctx.moveTo(f.x - 12 * f.facing, groundY - 45); ctx.lineTo(f.x + 15 * f.facing, groundY - 45); ctx.lineTo(f.x + 8 * f.facing, groundY - 105); ctx.lineTo(f.x - 18 * f.facing, groundY - 105); ctx.closePath(); ctx.fill(); ctx.stroke(); // Szorstka faktura rdzenia pancerza ctx.fillStyle = isFlashing ? SYSTĘM WYSUWANIA RAMION I ENERGETYCZNE TARCZE UDERZENIOWE let shoulderX = f.x - (2 * f.facing); let shoulderY = groundY - 95; let currentReach = 22; if (f.punching) { // Płynne, potężne wysunięcie teleskopowe na 100 pikseli do przodu currentReach = 22 + (f.punchProgress * 78); } let gloveX = shoulderX + (currentReach * f.facing); let gloveY = shoulderY - 8; // Ramię: Nowoczesna podwójna szyna mechaniczna zamiast zwykłej kreski ctx.strokeStyle = isFlashing ? "#ffffff" : f.color; ctx.lineWidth = 4; ctx.beginPath(); ctx.arc(gloveX, gloveY, 20, 0, Math.PI * 2); ctx.fill(); ctx.stroke(); // Wewnętrzny neonowy pierścień akceleratora ciosu ctx.strokeStyle = f.punching ? f.color : "rgba(100,100,100,0.5)"; ctx.lineWidth = 2; ctx.beginPath(); ctx.arc(gloveX, gloveY, 10, 0, Math.PI * 2); ctx.stroke(); ctx.restore(); } function AI(self, enemy) { self.timer--; self.facing = (enemy.x > self.x) ? 1 : -1; if (self.punching) return; // Blokada ruchu nóg podczas wyprostu let dist = Math.abs(enemy.x - self.x); if (self.timer <= 0) { let r = Math.random(); if (r < 0.50 && dist > 140) { // Natarcie: Doskok na precyzyjny zasięg uderzenia tarczy kinetycznej self.state = "forward"; self.targetX = enemy.x - (115 * self.facing); self.timer = 18; } else if (r < 0.88 && dist < 160) { // Unik taktyczny: Szybki mikrokrok w tył, by ramię wroga uderzyło w próżnię self.state = "backward"; self.targetX = self.x - (65 * self.facing); self.timer = 12; } else { // Dynamiczne wyczekiwanie w gardzie bocznej self.state = "idle"; self.timer = 8; } } if (self.state === "forward" || self.state === "backward") { let dx = self.targetX - self.x; if (Math.abs(dx) > 4) { self.x += (dx > 0 ? 1 : -1) * 3.0; // Prędkość poruszania się unowocześnionych botów } } // PRECYZYJNA DETEKCJA ODLEGŁOŚCI DLA WYPROSTU RAMIENIA if (!self.punching && dist >= 100 && dist <= 140 && Math.random() > 0.93) { self.punching = true; self.punchProgress = 0; self.punchStage = "out"; } } function handlePunchLogic(self, enemy) { if (!self.punching) return; if (self.punchStage === "out") { self.punchProgress += 0.18; // Bardziej eksplozywny, szybszy wyprost ramienia (Nowoczesna dynamika) if (self.punchProgress >= 1) { self.punchProgress = 1; self.punchStage = "return"; let shoulderX = self.x - (2 * self.facing); let myGloveGlobalX = shoulderX + (100 * self.facing); let enemyHeadX = enemy.x - (5 * enemy.facing); let hitDistance = Math.abs(enemyHeadX - myGloveGlobalX); // Trafienie wchodzi TYLKO jeśli środek tarczy kinetycznej uderzy centralnie w głowę przeciwnika if (hitDistance <= 25) { enemy.hp -= 10; if (enemy.hp < 0) enemy.hp = 0; enemy.hitFlash = 4; // Aktywacja błysku uszkodzeń na 4 klatki animacji document.getElementById("hp1").style.width = p1.hp + "%"; document.getElementById("hp2").style.width = p2.hp + "%"; logEvent(`KRACH KINETYCZNY! ${self.name} deklasuje obronę i trafia prosto w rdzeń głowy! (-10 HP)`, self.color); // Odrzut fizyczny oponenta po udanym trafieniu na wprost enemy.x -= (25 * enemy.facing); } else { logEvent(`${self.name} wyprowadził wyprost, ale wróg zrobił perfekcyjny spacing (Chybienie).`, "#666"); } } } else if (self.punchStage === "return") { self.punchProgress -= 0.12; // Szybki powrót ramienia do osłony bocznej if (self.punchProgress <= 0) { self.punchProgress = 0; self.punching = false; } } } function updateGame() { if (endOfGame) return; AI(p1, p2); AI(p2, p1); handlePunchLogic(p1, p2); handlePunchLogic(p2, p1); // Dynamiczne ściany ringu (Liny laserowe) [p1, p2].forEach(p => { p.x = Math.max(90, Math.min(610, p.x)); }); if (p1.hp <= 0 || p2.hp <= 0) { endOfGame = true; let winnerName = p1.hp > 0 ? p1.name : p2.name; let winnerColor = p1.hp > 0 ? p1.color : p2.color; logEvent(`SEKWENCJA ZAKOŃCZONA. DESTRUKCJA RDZENIA. Zwycięża: ${winnerName.toUpperCase()}!`, "#ffff00"); } } function drawScene() { ctx.clearRect(0, 0, canvas.width, canvas.height); // NOWOCZESNY CYBER-RING (Laserowe liny i cyfrowe podłoże) ctx.shadowBlur = 10; ctx.shadowColor = "#333333"; ctx.strokeStyle = "rgba(80, 80, 80, 0.4)"; ctx.lineWidth = 2; ctx.beginPath(); ctx.moveTo(40, 280); ctx.lineTo(660, 280); // Podłoga energetyczna ctx.stroke(); // Laserowe liny bokserskie w tle ctx.strokeStyle = "rgba(212, 175, 55, 0.25)"; ctx.lineWidth = 2; ctx.strokeRect(40, 160, 620, 1); ctx.strokeRect(40, 200, 620, 1); ctx.strokeRect(40, 240, 620, 1); // Konstrukcja słupów narożnikowych ctx.fillStyle = "#222"; ctx.fillRect(35, 130, 6, 150); ctx.fillRect(659, 130, 6, 150); // Rysowanie unowocześnionych zawodników drawModernFighter(p1); drawModernFighter(p2); } function gameLoop() { updateGame(); drawScene(); requestAnimationFrame(gameLoop); } logEvent("Gong cyfrowy!
