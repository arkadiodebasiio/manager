"); html.append(""); tadek.roundPoints = 10; damian.roundPoints = 10; for (int exchange = 1; exchange <= 5; exchange++) { if (tadek.isKnockedOut() || damian.isKnockedOut()) break; if (random.nextBoolean()) { simulateExchangeHtml(tadek, damian, html); } else { simulateExchangeHtml(damian, tadek, html); } } html.append("
Akcja w ringuStatus Zawodnikow
"); if (tadek.isKnockedOut() || damian.isKnockedOut()) break; tadek.totalScore += tadek.roundPoints; damian.totalScore += damian.roundPoints; tadek.regenerateStamina(); damian.regenerateStamina(); html.append("

Koniec rundy ").append(round).append(". Stan zdrowia -> ") .append(tadek.name).append(": ").append(tadek.hp).append("% HP | ") .append(damian.name).append(": ").append(damian.hp).append("% HP

Jednoglosna decyzja wygrywa: ").append(damian.getFullName()).append("!

Mamy remis!

"); // Zapisywanie calosci do pliku HTML, ktory mozesz otworzyc w przegladarce try (BufferedWriter writer = new BufferedWriter(new FileWriter("walka.html"))) { writer.write(html.toString()); System.out.println("SUKCES! Wygenerowano plik 'walka.html'. Otworz go w przegladarce, aby zobaczyc czytelny wynik!"); } catch (IOException e) { System.out.println("Blad zapisu pliku: " + e.getMessage()); } } private static void simulateExchangeHtml(Boxer attacker, Boxer defender, StringBuilder html) { PunchType punch = PunchType.values()[random.nextInt(PunchType.values().length)]; HitZone zone = HitZone.values()[random.nextInt(HitZone.values().length)]; attacker.stamina = Math.max(10, attacker.stamina - (8 + random.nextInt(7))); int attackRoll = attacker.speed + random.nextInt(40) + (attacker.stamina / 5); int defenseRoll = defender.defense + random.nextInt(40) + (defender.stamina / 5); if (attacker.stance != defender.stance) { attackRoll += 5; } html.append(""); if (attackRoll > defenseRoll) { int baseDamage = (attacker.power / 4) + random.nextInt(15); if (attacker.style == Style.SLUGGER && punch == PunchType.OVERHAND) baseDamage += 10; if (attacker.style == Style.OUT_BOXER && punch == PunchType.JAB) baseDamage += 5; int finalDamage = Math.max(2, baseDamage - (defender.chin / 15)); defender.hp = Math.max(0, defender.hp - finalDamage); html.append("[").append(attacker.nickname).append("] trafia potężnym ").append(punch).append(" na ").append(zone).append("!
