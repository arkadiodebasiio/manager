const canvas = document.getElementById('gameCanvas'); const ctx = canvas.getContext('2d'); const CONFIG = { speed: 5, jabDist: 80, hookDist: 60, staminaRegen: 0.6 }; // Dodane kolory rysowania bezpośrednio do obiektów zawodników const p1 = { id: 1, color: '#3388ff', gloveColor: '#1155cc', skinColor: '#ffdbac', hpEl: document.getElementById('p1-hp'), stamEl: document.getElementById('p1-stamina'), stateEl: document.getElementById('p1-state'), x: 180, hp: 100, stamina: 100, state: 'idle', stateTimer: 0 }; const p2 = { id: 2, color: '#ff3333', gloveColor: '#cc1111', skinColor: '#e0a96d', hpEl: document.getElementById('p2-hp'), stamEl: document.getElementById('p2-stamina'), stateEl: document.getElementById('p2-state'), x: 540, hp: 100, stamina: 100, state: 'idle', stateTimer: 0 }; const keys = {}; window.addEventListener('keydown', e => keys[e.code] = true); window.addEventListener('keyup', e => keys[e.code] = false); function punch(attacker, defender, type) { if (attacker.state !== 'idle' || attacker.stamina < 20) return; attacker.state = type; attacker.stateTimer = type === 'jab' ? 12 : 20; attacker.stamina -= type === 'jab' ? 15 : 25; const distance = Math.abs(attacker.x - defender.x); const reach = type === 'jab' ? CONFIG.jabDist : CONFIG.hookDist; const damage = type === 'jab' ? 7 : 16; if (distance <= reach + 70) { setTimeout(() => { if (defender.state === 'blocking') { defender.stamina = Math.max(0, defender.stamina - (damage / 2)); } else { defender.hp = Math.max(0, defender.hp - damage); defender.state = 'hit'; defender.stateTimer = 12; } }, 60); } } // CAŁKOWICIE NOWA FUNKCJA RYSOWANIA BOKSERÓW (Zastępuje puste klocki) function drawBoxer(p, facingRight) { ctx.save(); // Efekt wstrząsu/migania po otrzymaniu ciosu if (p.state === 'hit') { ctx.translate((Math.random() - 0.5) * 10, 0); ctx.filter = 'brightness(1.8)'; } const groundY = 355; // Wysokość podłogi ringu // Rysowanie Nóg / Spodenek bokserskich ctx.fillStyle = p.color; ctx.fillRect(p.x - 20, groundY - 40, 40, 40); ctx.fillStyle = '#111'; ctx.fillRect(p.x - 20, groundY, 15, -10); // Lewy but ctx.fillRect(p.x + 5, groundY, 15, -10); // Prawy but // Rysowanie Torsu (Klatka piersiowa) ctx.fillStyle = p.skinColor; ctx.fillRect(p.x - 25, groundY - 90, 50, 50); // Rysowanie Głowy ctx.beginPath(); ctx.arc(p.x, groundY - 110, 20, 0, Math.PI * 2); ctx.fillStyle = p.skinColor; ctx.fill(); ctx.closePath(); // Rysowanie Oczu/Włosów w zależności od kierunku patrzenia ctx.fillStyle = '#000'; const eyeOffset = facingRight ? 8 : -12; ctx.fillRect(p.x + eyeOffset, groundY - 115, 4, 4); // Oko // Rysowanie Rąk i Rękawic (Dynamiczne stany animacji) let glove1X = facingRight ? p.x + 15 : p.x - 35; let glove1Y = groundY - 75; let glove2X = facingRight ? p.x + 25 : p.x - 45; let glove2Y = groundY - 60; if (p.state === 'jab' || p.state === 'hook') { // Wyprowadzenie ciosu - rękawica leci do przodu const punchExtension = p.state === 'jab' ? CONFIG.jabDist : CONFIG.hookDist; glove2X = facingRight ? p.x + punchExtension : p.x - punchExtension; glove2Y = groundY - 80; } else if (p.state === 'blocking') { // Garda - rękawice zasłaniają twarz glove1X = facingRight ? p.x + 12 : p.x - 32; glove1Y = groundY - 110; glove2X = facingRight ? p.x + 20 : p.x - 40; glove2Y = groundY - 105; } // Rysowanie Rękawicy tylnej ctx.beginPath(); ctx.arc(glove1X, glove1Y, 11, 0, Math.PI * 2); ctx.fillStyle = p.gloveColor; ctx.fill(); ctx.strokeStyle = '#fff'; ctx.stroke(); // Rysowanie Rękawicy przedniej ctx.beginPath(); ctx.arc(glove2X, glove2Y, 13, 0, Math.PI * 2); ctx.fillStyle = p.gloveColor; ctx.fill(); ctx.stroke(); ctx.restore(); } function update() { [p1, p2].forEach(p => { if (p.stateTimer > 0) { p.stateTimer--; } else { if (p.state !== 'blocking') p.state = 'idle'; } if (p.state === 'idle' && p.stamina < 100) p.stamina = Math.min(100, p.stamina + CONFIG.staminaRegen); }); // Sterowanie P1 if (p1.state === 'idle' || p1.state === 'blocking') { if (keys['KeyW']) p1.state = 'blocking'; else if (p1.state === 'blocking') p1.state = 'idle'; if (p1.state === 'idle') { if (keys['KeyA']) p1.x = Math.max(25, p1.x - CONFIG.speed); if (keys['KeyD']) p1.x = Math.min(p2.x - 50, p1.x + CONFIG.speed); // Modyfikacja szerokości kolizji ciał if (keys['KeyG']) punch(p1, p2, 'jab'); if (keys['KeyH']) punch(p1, p2, 'hook'); } } // Sterowanie P2 if (p2.state === 'idle' || p2.state === 'blocking') { if (keys['ArrowUp']) p2.state = 'blocking'; else if (p2.state === 'blocking') p2.state = 'idle'; if (p2.state === 'idle') { if (keys['ArrowLeft']) p2.x = Math.max(p1.x + 50, p2.x - CONFIG.speed); if (keys['ArrowRight']) p2.x = Math.min(775, p2.x + CONFIG.speed); if (keys['Digit1'] || keys['Numpad1']) punch(p2, p1, 'jab'); if (keys['Digit2'] || keys['Numpad2']) punch(p2, p1, 'hook'); } } // CZYSZCZENIE I RENDEROWANIE SCENY RYSOWANEJ (Canvas) ctx.clearRect(0, 0, canvas.width, canvas.height); // Rysowanie lin i podłogi ringu z kodu ctx.fillStyle = '#3a3a5e'; ctx.fillRect(0, 350, canvas.width, 50); ctx.fillStyle = '#e94560'; ctx.fillRect(0, 345, canvas.width, 5); // Renderowanie wygenerowanych ciał bokserów drawBoxer(p1, true); // Gracz 1 patrzy w prawo drawBoxer(p2, false); // Gracz 2 patrzy w lewo // UI Aktualizacja barów p1.hpEl.style.width = p1.hp + '%'; p1.stamEl.style.width = p1.stamina + '%'; p1.stateEl.innerText = `STAN: ${p1.state.toUpperCase()}`; p2.hpEl.style.width = p2.hp + '%'; p2.stamEl.style.width = p2.stamina + '%'; p2.stateEl.innerText = `STAN: ${p2.state.toUpperCase()}`; if (p1.hp <= 0 || p2.hp <= 0) { alert(p1.hp <= 0 ? "K.O.! ROCKY WYGRYWA!" : "K.O.! MIGUEL WYGRYWA!"); p1.hp = 100; p2.hp = 100; p1.x = 180; p2.x = 540; p1.stamina = 100; p2.stamina = 100; } requestAnimationFrame(update); } update();
