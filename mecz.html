const canvas = document.getElementById("boxingRing"); const ctx = canvas.getContext("2d"); const commBox = document.getElementById("commentary"); function logEvent(text, color = "#00ff44") { commBox.innerHTML += `[RUNDA 1] ${text}
`; commBox.scrollTop = commBox.scrollHeight; } let p1 = { x: 150, baseRealY: 280, hp: 100, color: "#ff3333", name: "Czerwony", facing: 1, punching: false, punchProgress: 0, punchStage: "out", state: "move", timer: 0, targetX: 150 }; let p2 = { x: 550, baseRealY: 280, hp: 100, color: "#3388ff", name: "Niebieski", facing: -1, punching: false, punchProgress: 0, punchStage: "out", state: "move", timer: 0, targetX: 550 }; let endOfGame = false; function drawFighter(f) { ctx.save(); let groundY = f.baseRealY; // 1.
