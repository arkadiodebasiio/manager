const canvas = document.getElementById('ringCanvas'); const ctx = canvas.getContext('2d'); const CONFIG = { ringX: 80, ringY: 40, ringW: 640, ringH: 370, boxerRadius: 24 }; const p1 = { x: 240, y: 225, angle: 0, targetAngle: 0, color: '#00ddff', shortColor: '#0055aa', gloves: '#0099ff', hp: 100, punchState: 0, punchHand: 'left', speed: 3.5, activePunch: 'none' }; const p2 = { x: 560, y: 225, angle: Math.PI, targetAngle: Math.PI, color: '#ff3366', shortColor: '#aa0033', gloves: '#ff0055', hp: 100, punchState: 0, punchHand: 'left', speed: 3.5, activePunch: 'none' }; const keys = {}; window.addEventListener('keydown', e => keys[e.code] = true); window.addEventListener('keyup', e => keys[e.code] = false); function triggerPunch(attacker, type) { if (attacker.punchState > 0 || p1.hp <= 0 || p2.hp <= 0) return; attacker.activePunch = type; attacker.punchState = 1; attacker.punchHand = attacker.punchHand === 'left' ? 'right' : 'left'; } function processPunchLogic(attacker, defender) { if (attacker.punchState === 0) return; if (attacker.punchState < 6) { attacker.punchState += 1.5; if (attacker.punchState >= 5) { let reach = attacker.activePunch === 'jab' ? 30 : 20; let punchX = attacker.x + Math.cos(attacker.angle) * (CONFIG.boxerRadius + reach); let punchY = attacker.y + Math.sin(attacker.angle) * (CONFIG.boxerRadius + reach); let distToDefender = Math.hypot(punchX - defender.x, punchY - defender.y); if (distToDefender < CONFIG.boxerRadius + 5) { let damage = attacker.activePunch === 'jab' ? 6 : 14; defender.hp = Math.max(0, defender.hp - damage); createBloodSplatter(); } } } else { attacker.punchState += 1; if (attacker.punchState >= 11) { attacker.punchState = 0; attacker.activePunch = 'none'; } } } let flashIntensity = 0; function createBloodSplatter() { flashIntensity = 0.4; } function update() { // Ruch i celowanie P1 let p1MoveX = 0, p1MoveY = 0; if (keys['KeyW']) p1MoveY -= 1; if (keys['KeyS']) p1MoveY += 1; if (keys['KeyA']) p1MoveX -= 1; if (keys['KeyD']) p1MoveX += 1; if (p1MoveX !== 0 || p1MoveY !== 0) { p1.x += p1MoveX * p1.speed; p1.y += p1MoveY * p1.speed; p1.targetAngle = Math.atan2(p2.y - p1.y, p2.x - p1.x); } if (keys['KeyG']) triggerPunch(p1, 'jab'); if (keys['KeyH']) triggerPunch(p1, 'hook'); // Ruch i celowanie P2 let p2MoveX = 0, p2MoveY = 0; if (keys['ArrowUp']) p2MoveY -= 1; if (keys['ArrowDown']) p2MoveY += 1; if (keys['ArrowLeft']) p2MoveX -= 1; if (keys['ArrowRight']) p2MoveX += 1; if (p2MoveX !== 0 || p2MoveY !== 0) { p2.x += p2MoveX * p2.speed; p2.y += p2MoveY * p2.speed; p2.targetAngle = Math.atan2(p1.y - p2.y, p1.x - p2.x); } if (keys['Digit1'] || keys['Numpad1']) triggerPunch(p2, 'jab'); if (keys['Digit2'] || keys['Numpad2']) triggerPunch(p2, 'hook'); // Płynny obrót ramion i tułowia [p1, p2].forEach(p => { let diff = p.targetAngle - p.angle; while (diff < -Math.PI) diff += Math.PI * 2; while (diff > Math.PI) diff -= Math.PI * 2; p.angle += diff * 0.2; }); // Kolizje z linami ringu [p1, p2].forEach(p => { p.x = Math.max(CONFIG.ringX + CONFIG.boxerRadius, Math.min(CONFIG.ringX + CONFIG.ringW - CONFIG.boxerRadius, p.x)); p.y = Math.max(CONFIG.ringY + CONFIG.boxerRadius, Math.min(CONFIG.ringY + CONFIG.ringH - CONFIG.boxerRadius, p.y)); }); // Kolizja ciał graczy let dist = Math.hypot(p1.x - p2.x, p1.y - p2.y); let minDist = CONFIG.boxerRadius * 2; if (dist < minDist) { let overlap = minDist - dist; let dirX = (p1.x - p2.x) / dist; let dirY = (p1.y - p2.y) / dist; p1.x += dirX * overlap * 0.5; p1.y += dirY * overlap * 0.5; p2.x -= dirX * overlap * 0.5; p2.y -= dirY * overlap * 0.5; } processPunchLogic(p1, p2); processPunchLogic(p2, p1); document.getElementById('p1-hp-bar').style.width = p1.hp + '%'; document.getElementById('p2-hp-bar').style.width = p2.hp + '%'; if (flashIntensity > 0) flashIntensity -= 0.03; if (p1.hp <= 0 || p2.hp <= 0) { alert(p1.hp <= 0 ? "K.O.! ROCKY ROZBIJA RYWALA NA DESKI!" : "K.O.! MIGUEL WYGRYWA PRZEZ DESTRUKCJĘ!"); p1.hp = 100; p2.hp = 100; p1.x = 240; p2.x = 560; p1.angle = 0; p2.angle = Math.PI; } } function drawBoxer(p) { ctx.save(); ctx.translate(p.x, p.y); ctx.rotate(p.angle); // Cień pod zawodnikiem ctx.beginPath(); ctx.ellipse(0, 5, CONFIG.boxerRadius, CONFIG.boxerRadius - 6, 0, 0, Math.PI * 2); ctx.fillStyle = 'rgba(0, 0, 0, 0.4)'; ctx.fill(); // Tors i barki ctx.beginPath(); ctx.ellipse(0, 0, CONFIG.boxerRadius, CONFIG.boxerRadius - 4, 0, 0, Math.PI * 2); ctx.fillStyle = p.color; ctx.fill(); ctx.lineWidth = 3; ctx.strokeStyle = '#15161e'; ctx.stroke(); // Spodenki / Pas ctx.fillStyle = p.shortColor; ctx.fillRect(-CONFIG.boxerRadius + 6, -3, (CONFIG.boxerRadius - 6) * 2, 8); // Głowa ctx.beginPath(); ctx.arc(0, 0, 11, 0, Math.PI * 2); ctx.fillStyle = '#ffdbac'; ctx.fill(); ctx.stroke(); // Wskaźnik twarzy (nos) ctx.fillStyle = '#ffdbac'; ctx.fillRect(9, -2, 5, 4); // Pozycje rąk let leftHandDist = 12; let rightHandDist = 12; if (p.punchState > 0) { let extension = p.activePunch === 'jab' ? 24 : 16; let factor = Math.sin((p.punchState / 10) * Math.PI); if (p.punchHand === 'left') leftHandDist += extension * factor; if (p.punchHand === 'right') rightHandDist += extension * factor; } // Lewa rękawica ctx.beginPath(); ctx.arc(leftHandDist, -CONFIG.boxerRadius + 3, 8, 0, Math.PI * 2); ctx.fillStyle = p.gloves; ctx.fill(); ctx.stroke(); // Prawa rękawica ctx.beginPath(); ctx.arc(rightHandDist, CONFIG.boxerRadius - 3, 8, 0, Math.PI * 2); ctx.fillStyle = p.gloves; ctx.fill(); ctx.stroke(); ctx.restore(); } function draw() { ctx.fillStyle = '#222430'; ctx.fillRect(0, 0, canvas.width, canvas.height); ctx.fillStyle = '#1c1d27'; ctx.fillRect(CONFIG.ringX, CONFIG.ringY, CONFIG.ringW, CONFIG.ringH); ctx.strokeStyle = '#e94560'; ctx.lineWidth = 4; ctx.strokeRect(CONFIG.ringX, CONFIG.ringY, CONFIG.ringW, CONFIG.ringH); ctx.strokeStyle = '#ffffff'; ctx.lineWidth = 2; ctx.strokeRect(CONFIG.ringX + 4, CONFIG.ringY + 4, CONFIG.ringW - 8, CONFIG.ringH - 8); ctx.font = '800 70px Montserrat'; ctx.fillStyle = 'rgba(255, 255, 255, 0.02)'; ctx.textAlign = 'center'; ctx.fillText('BOXING', canvas.width / 2, canvas.height / 2 + 10); ctx.fillText('CHAMPION', canvas.width / 2, canvas.height / 2 + 75); ctx.fillStyle = '#ff3366'; ctx.fillRect(CONFIG.ringX - 6, CONFIG.ringY - 6, 12, 12); ctx.fillStyle = '#00ddff'; ctx.fillRect(CONFIG.ringX + CONFIG.ringW - 6, CONFIG.ringY - 6, 12, 12); ctx.fillStyle = '#fff'; ctx.fillRect(CONFIG.ringX - 6, CONFIG.ringY + CONFIG.ringH - 6, 12, 12); ctx.fillRect(CONFIG.ringX + CONFIG.ringW - 6, CONFIG.ringY + CONFIG.ringH - 6, 12, 12); if (flashIntensity > 0) { ctx.fillStyle = `rgba(255, 0, 85, ${flashIntensity})`; ctx.fillRect(CONFIG.ringX, CONFIG.ringY, CONFIG.ringW, CONFIG.ringH); } drawBoxer(p1); drawBoxer(p2); } function gameLoop() { update(); draw(); requestAnimationFrame(gameLoop); } gameLoop();
