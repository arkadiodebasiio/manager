// Linki do stanów grafik (Sprite'y retro) const SPRITES = { idle: 'https://discourse-cdn.com', punch: 'https://squarespace-cdn.com', block: 'https://openclipart.org' }; const CONFIG = { speed: 5, jabDist: 70, hookDist: 50, staminaRegen: 0.6 }; const p1 = { id: 1, el: document.getElementById('player1'), hpEl: document.getElementById('p1-hp'), stamEl: document.getElementById('p1-stamina'), stateEl: document.getElementById('p1-state'), x: 180, hp: 100, stamina: 100, state: 'idle', stateTimer: 0 }; const p2 = { id: 2, el: document.getElementById('player2'), hpEl: document.getElementById('p2-hp'), stamEl: document.getElementById('p2-stamina'), stateEl: document.getElementById('p2-state'), x: 510, hp: 100, stamina: 100, state: 'idle', stateTimer: 0 }; const keys = {}; window.addEventListener('keydown', e => keys[e.code] = true); window.addEventListener('keyup', e => keys[e.code] = false); function punch(attacker, defender, type) { if (attacker.state !== 'idle' || attacker.stamina < 20) return; attacker.state = type; attacker.stateTimer = type === 'jab' ? 12 : 20; attacker.stamina -= type === 'jab' ? 15 : 25; const distance = Math.abs(attacker.x - defender.x); const reach = type === 'jab' ? CONFIG.jabDist : CONFIG.hookDist; const damage = type === 'jab' ? 7 : 16; if (distance <= reach + 80) { setTimeout(() => { if (defender.state === 'blocking') { defender.stamina = Math.max(0, defender.stamina - (damage / 2)); } else { defender.hp = Math.max(0, defender.hp - damage); defender.state = 'hit'; defender.stateTimer = 10; } }, 60); } } function drawPlayer(player, isP2 = false) { player.el.style.left = player.x + 'px'; if (player.state === 'jab' || player.state === 'hook') { player.el.style.backgroundImage = `url('${SPRITES.punch}')`; } else if (player.state === 'blocking') { player.el.style.backgroundImage = `url('${SPRITES.block}')`; } else { player.el.style.backgroundImage = `url('${SPRITES.idle}')`; } if (player.state === 'hit') { player.el.classList.add('hit-effect'); } else { player.el.classList.remove('hit-effect'); } if (isP2) { player.el.style.transform = "scaleX(-1)"; } } function update() { [p1, p2].forEach(p => { if (p.stateTimer > 0) { p.stateTimer--; } else { if (p.state !== 'blocking') p.state = 'idle'; } if (p.state === 'idle' && p.stamina < 100) p.stamina = Math.min(100, p.stamina + CONFIG.staminaRegen); }); // Sterowanie Gracz 1 if (p1.state === 'idle' || p1.state === 'blocking') { if (keys['KeyW']) p1.state = 'blocking'; else if (p1.state === 'blocking') p1.state = 'idle'; if (p1.state === 'idle') { if (keys['KeyA']) p1.x = Math.max(0, p1.x - CONFIG.speed); if (keys['KeyD']) p1.x = Math.min(p2.x - 70, p1.x + CONFIG.speed); if (keys['KeyG']) punch(p1, p2, 'jab'); if (keys['KeyH']) punch(p1, p2, 'hook'); } } // Sterowanie Gracz 2 if (p2.state === 'idle' || p2.state === 'blocking') { if (keys['ArrowUp']) p2.state = 'blocking'; else if (p2.state === 'blocking') p2.state = 'idle'; if (p2.state === 'idle') { if (keys['ArrowLeft']) p2.x = Math.max(p1.x + 70, p2.x - CONFIG.speed); if (keys['ArrowRight']) p2.x = Math.min(690, p2.x + CONFIG.speed); if (keys['Digit1'] || keys['Numpad1']) punch(p2, p1, 'jab'); if (keys['Digit2'] || keys['Numpad2']) punch(p2, p1, 'hook'); } } drawPlayer(p1, false); drawPlayer(p2, true); p1.hpEl.style.width = p1.hp + '%'; p1.stamEl.style.width = p1.stamina + '%'; p1.stateEl.innerText = `STAN: ${p1.state.toUpperCase()}`; p2.hpEl.style.width = p2.hp + '%'; p2.stamEl.style.width = p2.stamina + '%'; p2.stateEl.innerText = `STAN: ${p2.state.toUpperCase()}`; if (p1.hp <= 0 || p2.hp <= 0) { alert(p1.hp <= 0 ? "K.O.! ROCKY (GRACZ 2) WYGRYWA!" : "K.O.! MIGUEL (GRACZ 1) WYGRYWA!"); p1.hp = 100; p2.hp = 100; p1.x = 180; p2.x = 510; p1.stamina = 100; p2.stamina = 100; } requestAnimationFrame(update); } // Start pętli gry po załadowaniu skryptu update();
