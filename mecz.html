import java.util.Random; import java.util.concurrent.TimeUnit; public class BoxingMatch { private static final Random random = new Random(); enum Stance { ORTHODOX, SOUTHPAW } enum Style { SLUGGER, OUT_BOXER, INFIGHTER } enum HitZone { HEAD, BODY } enum PunchType { JAB, HOOK, UPPERCUT, OVERHAND } static class Boxer { String name; String nickname; Stance stance; Style style; int hp = 100; int stamina = 100; int power; int speed; int chin; int defense; int roundPoints = 0; int totalScore = 0; public Boxer(String name, String nickname, Stance stance, Style style, int power, int speed, int chin, int defense) { this.name = name; this.nickname = nickname; this.stance = stance; this.style = style; this.power = power; this.speed = speed; this.chin = chin; this.defense = defense; } public String getFullName() { return name + " \"" + nickname + "\""; } public boolean isKnockedOut() { return this.hp <= 0; } public void regenerateStamina() { this.stamina = Math.min(100, this.stamina + 25); } } public static void main(String[] args) { // Usunieto polskie znaki z nazwisk dla bezpieczenstwa kodowania stron www Boxer tadek = new Boxer("Tomasz Majewski", "Grom", Stance.SOUTHPAW, Style.SLUGGER, 88, 70, 90, 85); Boxer damian = new Boxer("Damian Orlowski", "Orzel", Stance.ORTHODOX, Style.OUT_BOXER, 75, 92, 78, 80); System.out.println("=== WALKA O PAS MISTRZOWSKI WAGI CIEZKIEJ ===\n"); System.out.println(tadek.getFullName() + " VS " + damian.getFullName() + "\n"); System.out.println("---------------------------------------------------\n"); int totalRounds = 3; for (int round = 1; round <= totalRounds; round++) { System.out.println("--- RUNDA " + round + " ---\n"); tadek.roundPoints = 10; damian.roundPoints = 10; for (int exchange = 1; exchange <= 5; exchange++) { if (tadek.isKnockedOut() || damian.isKnockedOut()) break; if (random.nextBoolean()) { simulateExchange(tadek, damian); } else { simulateExchange(damian, tadek); } sleep(1000); // Zwiekszono pauze, aby tekst nie "lecial" za szybko } if (tadek.isKnockedOut() || damian.isKnockedOut()) break; System.out.println("\n>> Koniec rundy " + round + ". Zawodnicy wracaja do naroznikow. <<\n"); tadek.totalScore += tadek.roundPoints; damian.totalScore += damian.roundPoints; tadek.regenerateStamina(); damian.regenerateStamina(); System.out.printf("Stan zdrowia -> %s: %d%% HP | %s: %d%% HP\n", tadek.name, tadek.hp, damian.name, damian.hp); System.out.println("---------------------------------------------------\n"); sleep(1500); } System.out.println("\n================= WERDYKT WALKI ================="); if (tadek.isKnockedOut()) { System.out.println("KONIEC PRZED CZASEM! KO! Zwyciezca zostaje: " + damian.getFullName() + "!"); } else if (damian.isKnockedOut()) { System.out.println("KONIEC PRZED CZASEM! KO! Zwyciezca zostaje: " + tadek.getFullName() + "!"); } else { System.out.println("Walka dotrwala do decyzji sedziow!"); System.out.printf("Punkty sedziowskie: %s [%d] vs [%d] %s\n", tadek.name, tadek.totalScore, damian.totalScore, damian.name); if (tadek.totalScore > damian.totalScore) { System.out.println("Jednoglosna decyzja sedziow wygrywa: " + tadek.getFullName() + "!"); } else if (damian.totalScore > tadek.totalScore) { System.out.println("Jednoglosna decyzja sedziow wygrywa: " + damian.getFullName() + "!"); } else { System.out.println("Mamy remis!"); } } } private static void simulateExchange(Boxer attacker, Boxer defender) { PunchType punch = PunchType.values()[random.nextInt(PunchType.values().length)]; HitZone zone = HitZone.values()[random.nextInt(HitZone.values().length)]; attacker.stamina = Math.max(10, attacker.stamina - (8 + random.nextInt(7))); int attackRoll = attacker.speed + random.nextInt(40) + (attacker.stamina / 5); int defenseRoll = defender.defense + random.nextInt(40) + (defender.stamina / 5); if (attacker.stance != defender.stance) { attackRoll += 5; } // Kazdy podpunkt logowania konczy sie teraz jawnym znakiem nowej linii \n if (attackRoll > defenseRoll) { int baseDamage = (attacker.power / 4) + random.nextInt(15); if (attacker.style == Style.SLUGGER && punch == PunchType.OVERHAND) baseDamage += 10; if (attacker.style == Style.OUT_BOXER && punch == PunchType.JAB) baseDamage += 5; int finalDamage = Math.max(2, baseDamage - (defender.chin / 15)); defender.hp = Math.max(0, defender.hp - finalDamage); System.out.printf("[%s] trafia poteznym %s na %s! (%d obrazen)\n", attacker.nickname, punch, zone, finalDamage); if (finalDamage > 18 && random.nextInt(100) < 25) { System.out.printf(" !!! ALERT: %s laduje na deskach! Sedzia zaczyna liczenie! !!!\n", defender.name); defender.roundPoints -= 2; defender.stamina = Math.max(20, defender.stamina - 30); if (defender.hp <= 10 && random.nextInt(100) > defender.chin) { defender.hp = 0; System.out.printf("...Osiem... Dziewiec... Dziesiec! Koniec! %s zostaje znokautowany!\n", defender.name); } else { System.out.println("...Bokser podnosi sie na 'osiem' i walczy dalej!\n"); } } } else if (defenseRoll - attackRoll < 10) { System.out.printf("[%s] blokuje cios, klincz! Sedzia rozrywa pozycje.\n", defender.nickname); } else { System.out.printf("[%s] robi genialny unik! %s przestrzeliwuje.\n", defender.nickname, attacker.nickname); } } private static void sleep(int millis) { try { TimeUnit.MILLISECONDS.sleep(millis); } catch (InterruptedException e) { Thread.currentThread().interrupt(); } } }
